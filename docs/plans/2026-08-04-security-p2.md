pub const HKDF_INFO: &[u8] = b"ctp-secrets-v2";

pub struct KeyRing {
    pub current: String,
    pub previous: Option<String>,
    pub current_version: i32, // typically 2
}

fn derive_key_v1(secret: &str) -> [u8; 32] { /* existing Sha256 */ }

fn derive_key_v2(secret: &str) -> [u8; 32] {
    // HKDF-SHA256, salt empty or b"ctp", info = HKDF_INFO, length 32
}

pub fn encrypt_secret_versioned(
    plaintext: &[u8],
    ring: &KeyRing,
) -> Result<(Vec<u8>, Vec<u8>, i32), CryptoError> {
    let ver = ring.current_version.max(2); // new writes use HKDF path
    let key = if ver <= 1 { derive_key_v1(&ring.current) } else { derive_key_v2(&ring.current) };
    // AES-GCM as today → (nonce, ct, ver)
}

pub fn decrypt_secret_versioned(
    nonce: &[u8],
    ciphertext: &[u8],
    key_version: i32,
    ring: &KeyRing,
) -> Result<String, CryptoError> {
    let materials: Vec<&str> = std::iter::once(ring.current.as_str())
        .chain(ring.previous.as_deref())
        .collect();
    for secret in materials {
        let key = if key_version <= 1 {
            derive_key_v1(secret)
        } else {
            derive_key_v2(secret)
        };
        if let Ok(s) = try_decrypt(nonce, ciphertext, &key) {
            return Ok(s);
        }
    }
    Err(CryptoError::Decrypt)
}

### Task 11: CI script + GitHub Actions

**Files:**
- Create: `scripts/ci-security.sh` (executable)
- Create: `.github/workflows/security.yml`
- Optional: `deny.toml` only if using cargo-deny; prefer `cargo audit` for simplicity

**`scripts/ci-security.sh`:**

```bash
#!/usr/bin/env bash
set -euo pipefail
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
cd "$ROOT"

echo "==> cargo fetch"
cargo fetch

if command -v cargo-audit >/dev/null 2>&1 || cargo audit -V >/dev/null 2>&1; then
  cargo audit
else
  echo "cargo-audit not installed; install: cargo install cargo-audit --locked"
  # In CI the workflow installs it; locally warn but exit 1 if STRICT=1
  if [[ "${STRICT:-}" == "1" ]]; then exit 1; fi
fi

if command -v trivy >/dev/null 2>&1; then
  trivy fs --severity HIGH,CRITICAL --exit-code 1 .
  if [[ -f Dockerfile ]]; then
    trivy config Dockerfile || true
  fi
else
  echo "trivy not installed; skipping image/fs scan"
  if [[ "${STRICT:-}" == "1" && "${REQUIRE_TRIVY:-}" == "1" ]]; then exit 1; fi
fi
```

**Workflow:**

- on: pull_request, push main, schedule weekly  
- job rust-audit: install cargo-audit, `cargo audit`  
- job trivy: `aquasecurity/trivy-action` on fs + Dockerfile  

**Commit:** `ci: security workflow and local ci-security.sh`
